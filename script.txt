--[[
    ULTRA MINIMAL GIFT ESP (NAME ONLY)
    - Auto-scanning: Always On
    - No Menu, No Lag
]]

local RunService = game:GetService("RunService")
local TargetKeyword = "Gift" -- Từ khóa để quét

-- Hàm dọn dẹp các tag cũ để tránh lỗi chồng chéo
local function ClearOldTags()
    for _, v in pairs(workspace:GetDescendants()) do
        if v.Name == "GiftTag" then
            v:Destroy()
        end
    end
end

-- Dọn dẹp một lần khi mới chạy script
ClearOldTags()

-- Vòng lặp quét liên tục mỗi giây
RunService.Heartbeat:Connect(function()
    for _, v in pairs(workspace:GetDescendants()) do
        -- Kiểm tra nếu là BasePart và tên bắt đầu bằng "Gift"
        if v:IsA("BasePart") and v.Name:sub(1, #TargetKeyword) == TargetKeyword then
            
            -- Nếu chưa có Tag thì tạo mới
            if not v:FindFirstChild("GiftTag") then
                local bg = Instance.new("BillboardGui")
                bg.Name = "GiftTag"
                bg.Parent = v
                bg.AlwaysOnTop = true
                bg.Size = UDim2.new(0, 100, 0, 20)
                bg.ExtentsOffset = Vector3.new(0, 3, 0) -- Độ cao của chữ trên vật phẩm

                local lbl = Instance.new("TextLabel")
                lbl.Parent = bg
                lbl.Size = UDim2.new(1, 0, 1, 0)
                lbl.BackgroundTransparency = 1
                lbl.Text = "🎁 GIFT" -- Hiển thị tên
                lbl.TextColor3 = Color3.fromRGB(255, 255, 255) -- Màu trắng
                lbl.TextStrokeTransparency = 0 -- Viền đen cho dễ nhìn
                lbl.TextStrokeColor3 = Color3.fromRGB(0, 0, 0)
                lbl.Font = Enum.Font.GothamBold
                lbl.TextSize = 14
            end
        end
    end
end)

print("Gift ESP (Name Only) đã hoạt động!")
