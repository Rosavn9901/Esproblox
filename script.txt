local Players = game:GetService("Players")
local LP = Players.LocalPlayer
local RunService = game:GetService("RunService")

local MAX_DIST = 300
local GiftKey = "Gift"

local function CreateTag(parent, name, color)
    if parent:FindFirstChild(name) then return parent[name] end
    local bg = Instance.new("BillboardGui", parent)
    bg.Name = name; bg.AlwaysOnTop = true; bg.Size = UDim2.new(0, 100, 0, 40); bg.ExtentsOffset = Vector3.new(0, 3, 0)
    local m = Instance.new("TextLabel", bg); m.Name = "Main"; m.Size = UDim2.new(1, 0, 0.5, 0); m.BackgroundTransparency = 1; m.Font = "GothamBold"; m.TextSize = 12; m.TextStrokeTransparency = 0
    local a = Instance.new("TextLabel", bg); a.Name = "Arrow"; a.Size = UDim2.new(1, 0, 0.5, 0); a.Position = UDim2.new(0,0,0.5,0); a.BackgroundTransparency = 1; a.Text = "▼"; a.TextSize = 14; a.TextStrokeTransparency = 0
    m.TextColor3 = color; a.TextColor3 = color
    return bg
end

RunService.RenderStepped:Connect(function()
    local myRoot = LP.Character and LP.Character:FindFirstChild("HumanoidRootPart")
    if not myRoot then return end

    for _, v in pairs(workspace:GetDescendants()) do
        if v:IsA("BasePart") and v.Name:find(GiftKey) and not v.Name:lower():find("fake") then
            local target = v:FindFirstAncestorOfClass("Model") or v
            local dist = (v.Position - myRoot.Position).Magnitude
            
            if dist <= MAX_DIST then
                local tag = CreateTag(target, "GiftTag", Color3.new(1, 1, 0))
                tag.Enabled = true
                tag.Main.Text = "🎁 GIFT [" .. math.floor(dist) .. "m]"
            elseif target:FindFirstChild("GiftTag") then
                target.GiftTag.Enabled = false
            end
        end
    end
    
    for _, v in pairs(workspace:GetDescendants()) do
        if v.Name == "GiftTag" and (not v.Parent or not v.Parent:IsDescendantOf(workspace)) then
            v:Destroy()
        end
    end
end)
