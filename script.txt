--[[
    OPTIMIZED ESP V10 (ULTRA LAG FIX)
    - Optimization: Throttled loop (0.1s) to save CPU.
    - Logic: Smart filtering to prevent duplicate tags.
    - Performance: Reduced workspace scanning.
]]

local Players = game:GetService("Players")
local LP = Players.LocalPlayer
local MAX_DISTANCE = 300 
local TargetKeyword = "Gift"
local UPDATE_INTERVAL = 0.1 -- Cập nhật sau mỗi 0.1 giây (giảm lag cực tốt)

-- Hàm lấy màu Team
local function GetPlayerColor(p)
    if p.TeamColor then return p.TeamColor.Color end
    return Color3.new(1, 1, 1)
end

-- Vòng lặp tối ưu
task.spawn(function()
    while task.wait(UPDATE_INTERVAL) do
        local character = LP.Character
        local rootPart = character and character:FindFirstChild("HumanoidRootPart")
        if not rootPart then continue end

        -- 1. XỬ LÝ GIFT ESP (Tối ưu quét)
        for _, v in pairs(workspace:GetChildren()) do -- Chỉ quét lớp ngoài để nhanh hơn
            -- Nếu Gift nằm trong Folder hoặc Model, bạn có thể đổi thành workspace:GetDescendants() nếu cần
            if v.Name:sub(1, #TargetKeyword) == TargetKeyword and v:IsA("BasePart") then
                local dist = (v.Position - rootPart.Position).Magnitude
                
                if dist <= MAX_DISTANCE then
                    local bg = v:FindFirstChild("GiftTag")
                    if not bg then
                        bg = Instance.new("BillboardGui", v)
                        bg.Name = "GiftTag"
                        bg.AlwaysOnTop = true
                        bg.Size = UDim2.new(0, 100, 0, 40)
                        bg.ExtentsOffset = Vector3.new(0, 3, 0)

                        local label = Instance.new("TextLabel", bg)
                        label.Name = "Main"
                        label.Size = UDim2.new(1, 0, 0.5, 0)
                        label.BackgroundTransparency = 1
                        label.TextColor3 = Color3.fromRGB(255, 255, 0)
                        label.TextStrokeTransparency = 0
                        label.Font = Enum.Font.GothamBold
                        label.TextSize = 11

                        local arrow = Instance.new("TextLabel", bg)
                        arrow.Name = "Arrow"
                        arrow.Size = UDim2.new(1, 0, 0.5, 0)
                        arrow.Position = UDim2.new(0, 0, 0.5, 0)
                        arrow.BackgroundTransparency = 1
                        arrow.Text = "▼"
                        arrow.TextColor3 = Color3.new(1, 1, 1)
                        arrow.TextStrokeTransparency = 0
                        arrow.TextSize = 13
                    end
                    bg.Main.Text = "🎁 GIFT [" .. math.floor(dist) .. "m]"
                else
                    if v:FindFirstChild("GiftTag") then v.GiftTag:Destroy() end
                end
            end
        end

        -- 2. XỬ LÝ PLAYER ESP (Tối ưu quét)
        for _, p in pairs(Players:GetPlayers()) do
            if p ~= LP and p.Character and p.Character:FindFirstChild("HumanoidRootPart") then
                local pRoot = p.Character.HumanoidRootPart
                local dist = (pRoot.Position - rootPart.Position).Magnitude
                
                if dist <= MAX_DISTANCE then
                    local teamColor = GetPlayerColor(p)
                    local bg = pRoot:FindFirstChild("PlayerTag")
                    if not bg then
                        bg = Instance.new("BillboardGui", pRoot)
                        bg.Name = "PlayerTag"
                        bg.AlwaysOnTop = true
                        bg.Size = UDim2.new(0, 120, 0, 40)
                        bg.ExtentsOffset = Vector3.new(0, 4, 0)

                        local label = Instance.new("TextLabel", bg)
                        label.Name = "Main"
                        label.Size = UDim2.new(1, 0, 0.5, 0)
                        label.BackgroundTransparency = 1
                        label.TextColor3 = teamColor
                        label.TextStrokeTransparency = 0
                        label.Font = Enum.Font.GothamBold
                        label.TextSize = 11

                        local arrow = Instance.new("TextLabel", bg)
                        arrow.Name = "Arrow"
                        arrow.Size = UDim2.new(1, 0, 0.5, 0)
                        arrow.Position = UDim2.new(0, 0, 0.5, 0)
                        arrow.BackgroundTransparency = 1
                        arrow.Text = "▼"
                        arrow.TextColor3 = teamColor
                        arrow.TextStrokeTransparency = 0
                        arrow.TextSize = 13
                    end
                    bg.Main.Text = p.Name .. " [" .. math.floor(dist) .. "m]"
                    bg.Main.TextColor3 = teamColor
                    bg.Arrow.TextColor3 = teamColor
                else
                    if pRoot:FindFirstChild("PlayerTag") then pRoot.PlayerTag:Destroy() end
                end
            end
        end
    end
end)

print("V10 Ultra Lag Fix đã sẵn sàng!")
