local function GetRawName(name) return name:match("^(.-)%s*%d*$") or name end

local function SaveConfig() if writefile then pcall(function() writefile(ConfigFile, HttpService:JSONEncode(Config)) end) end end
local function LoadConfig()
    if isfile and isfile(ConfigFile) then
        local s, d = pcall(function() return HttpService:JSONDecode(readfile(ConfigFile)) end)
        if s then Config = d end
    end
end
LoadConfig()

local function ClearESP(mode)
    if mode == "Item" or mode == "All" then
        for _, v in pairs(workspace:GetDescendants()) do
            if v:FindFirstChild("ESPTag") then v.ESPTag:Destroy() end
            if v:FindFirstChild("ChamTag") then v.ChamTag:Destroy() end
        end
    end
    if mode == "Player" or mode == "All" then
        for _, p in pairs(Players:GetPlayers()) do
            if p.Character then
                for _, x in pairs(p.Character:GetDescendants()) do 
                    if x.Name == "PlayerESP" or x.Name == "PlayerCham" then x:Destroy() end 
                end
            end
        end
    end
end

-- 1. HÀM DRAGGABLE (MOBILE + PC)
local function MakeDraggable(guiObj)
    local dragging, dragInput, dragStart, startPos
    local function update(input)
        local delta = input.Position - dragStart
        guiObj.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
    end
    guiObj.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true; dragStart = input.Position; startPos = guiObj.Position
            input.Changed:Connect(function() if input.UserInputState == Enum.UserInputState.End then dragging = false end end)
        end
    end)
    guiObj.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then dragInput = input end
    end)
    UserInputService.InputChanged:Connect(function(input) if input == dragInput and dragging then update(input) end end)
end

-- === LOGIC ESP CORE (PLAYER + ITEM) ===
local function CreatePlayerESP(p)
    if not Config.States.PlayerESP then return end
    if p == LP or not p.Character then return end
    
    local char = p.Character
    local hrp = char:WaitForChild("HumanoidRootPart", 1) or char:WaitForChild("Torso", 1)
    if not hrp then return end

    local isAlly = (p.Team ~= nil and p.Team == LP.Team)
    local color = isAlly and Color3.new(0, 1, 0) or Color3.new(1, 0, 0)

    if not char:FindFirstChild("PlayerCham") then
        local hl = Instance.new("Highlight", char); hl.Name = "PlayerCham"; hl.FillColor = color; hl.FillTransparency = 0.5; hl.OutlineColor = Color3.new(1,1,1)
    end
    if not hrp:FindFirstChild("PlayerESP") then
        local bg = Instance.new("BillboardGui", hrp); bg.Name = "PlayerESP"; bg.AlwaysOnTop = true; bg.Size = UDim2.new(0,100,0,20); bg.ExtentsOffset = Vector3.new(0,3,0)
        local lbl = Instance.new("TextLabel", bg); lbl.Size = UDim2.new(1,0,1,0); lbl.BackgroundTransparency = 1; lbl.Text = p.Name; lbl.TextColor3 = color; lbl.TextStrokeTransparency = 0; lbl.Font = Enum.Font.GothamBold; lbl.TextSize = 12
    end
end
local function ConnectPlayer(p)
    p.CharacterAdded:Connect(function() task.wait(1); CreatePlayerESP(p) end)
    if p.Character then CreatePlayerESP(p) end
end
for _, p in pairs(Players:GetPlayers()) do ConnectPlayer(p) end
Players.PlayerAdded:Connect(ConnectPlayer)

-- Auto Path Loop
local moving = false
local function MoveToItem(target)
    if moving or not target or not target.Parent then return end
    moving = true
    local hum = LP.Character and LP.Character:FindFirstChild("Humanoid")
    if hum then
        local path = PathfindingService:CreatePath({AgentRadius=2, AgentCanJump=true})
        pcall(function()
            path:ComputeAsync(LP.Character.HumanoidRootPart.Position, target.Position)
            if path.Status == Enum.PathStatus.Success then
                for _, wp in pairs(path:GetWaypoints()) do
                    if not Config.States.AutoPath or not target.Parent then break end
                    if wp.Action == Enum.PathWaypointAction.Jump then hum.Jump = true end
                    hum:MoveTo(wp.Position); hum.MoveToFinished:Wait()
                end
            else hum:MoveTo(target.Position) end
        end)
    end
    moving = false
end

RunService.Heartbeat:Connect(function()
    if Config.States.PlayerESP then
        for _, p in pairs(Players:GetPlayers()) do if p ~= LP and p.Character then CreatePlayerESP(p) end end
    end

    local nearestItem, minDst = nil, math.huge
    for _, v in pairs(workspace:GetDescendants()) do
        if v:IsA("BasePart") then
            local key = nil
            for _, k in pairs(Config.Items) do if v.Name:sub(1, #k) == k then key = k; break end end
            
            if key then
                local dst = LP.Character and LP.Character:FindFirstChild("HumanoidRootPart") and (v.Position - LP.Character.HumanoidRootPart.Position).Magnitude or 9999
                if dst < minDst then minDst = dst; nearestItem = v end

                if Config.States.ESP then
                    local bg = v:FindFirstChild("ESPTag") or Instance.new("BillboardGui", v); bg.Name = "ESPTag"; bg.AlwaysOnTop = true; bg.Size = UDim2.new(0,80,0,20); bg.ExtentsOffset = Vector3.new(0,2,0)
                    local lbl = bg:FindFirstChild("T") or Instance.new("TextLabel", bg); lbl.Name = "T"; lbl.Size = UDim2.new(1,0,1,0); lbl.BackgroundTransparency = 1; lbl.Text = key; lbl.TextColor3 = Color3.new(1,1,1); lbl.TextStrokeTransparency = 0; lbl.Font = Enum.Font.GothamBold; lbl.TextSize = 11
                elseif v:FindFirstChild("ESPTag") then v.ESPTag:Destroy() end

                if Config.States.Chams then
                    local hl = v:FindFirstChild("ChamTag") or Instance.new("Highlight", v); hl.Name = "ChamTag"; hl.FillColor = Color3.new(1, 1, 0); hl.OutlineColor = Color3.new(1,1,1)
                elseif v:FindFirstChild("ChamTag") then v.ChamTag:Destroy() end
            end
        end
    end
    if Config.States.AutoPath and nearestItem then task.spawn(function() MoveToItem(nearestItem) end) end
end)
