--[[
    PLAYER ESP LAG-FREE VERSION
    - Logic: Event-based (No Workspace Scanning).
    - Performance: Extremely Low CPU Usage.
    - Range: Infinite (Hoặc tùy chỉnh).
]]

local Players = game:GetService("Players")
local LP = Players.LocalPlayer
local RunService = game:GetService("RunService")

-- Cấu hình
local SHOW_DISTANCE = true
local TargetKeyword = "Gift" -- Vẫn giữ Gift nhưng sẽ dùng tối ưu riêng

-- Hàm lấy màu Team
local function GetPlayerColor(p)
    if p.TeamColor then return p.TeamColor.Color end
    return Color3.new(1, 1, 1)
end

-- HÀM TẠO ESP CHO NGƯỜI CHƠI (Chỉ chạy 1 lần khi người chơi load)
local function CreatePlayerESP(p)
    p.CharacterAdded:Connect(function(char)
        local root = char:WaitForChild("HumanoidRootPart", 10)
        if not root then return end

        local bg = root:FindFirstChild("PlayerTag") or Instance.new("BillboardGui")
        bg.Name = "PlayerTag"
        bg.Parent = root
        bg.AlwaysOnTop = true
        bg.Size = UDim2.new(0, 120, 0, 40)
        bg.ExtentsOffset = Vector3.new(0, 4, 0)

        local label = bg:FindFirstChild("Main") or Instance.new("TextLabel", bg)
        label.Name = "Main"
        label.Size = UDim2.new(1, 0, 0.5, 0)
        label.BackgroundTransparency = 1
        label.Font = Enum.Font.GothamBold
        label.TextSize = 11
        label.TextStrokeTransparency = 0

        local arrow = bg:FindFirstChild("Arrow") or Instance.new("TextLabel", bg)
        arrow.Name = "Arrow"
        arrow.Size = UDim2.new(1, 0, 0.5, 0)
        arrow.Position = UDim2.new(0, 0, 0.5, 0)
        arrow.BackgroundTransparency = 1
        arrow.Text = "▼"
        arrow.TextStrokeTransparency = 0
        arrow.TextSize = 13
    end)
end

-- Khởi tạo cho những người đã có sẵn trong server
for _, p in pairs(Players:GetPlayers()) do
    if p ~= LP then
        CreatePlayerESP(p)
        if p.Character then -- Nếu họ đã load nhân vật rồi
            -- Gọi thủ công CharacterAdded cho họ
            task.spawn(function()
                local char = p.Character
                local root = char:FindFirstChild("HumanoidRootPart")
                if root then
                    -- Tự kích hoạt lại logic tạo nhãn nếu cần
                end
            end)
        end
    end
end

-- Lắng nghe người chơi mới vào
Players.PlayerAdded:Connect(function(p)
    if p ~= LP then CreatePlayerESP(p) end
end)

-- VÒNG LẶP NHẸ (Chỉ cập nhật dữ liệu, không quét vật thể)
RunService.RenderStepped:Connect(function()
    local myRoot = LP.Character and LP.Character:FindFirstChild("HumanoidRootPart")
    if not myRoot then return end

    -- Cập nhật Player ESP
    for _, p in pairs(Players:GetPlayers()) do
        if p ~= LP and p.Character then
            local pRoot = p.Character:FindFirstChild("HumanoidRootPart")
            if pRoot and pRoot:FindFirstChild("PlayerTag") then
                local tag = pRoot.PlayerTag
                local dist = (pRoot.Position - myRoot.Position).Magnitude
                local color = GetPlayerColor(p)
                
                tag.Main.Text = p.Name .. " [" .. math.floor(dist) .. "m]"
                tag.Main.TextColor3 = color
                tag.Arrow.TextColor3 = color
            end
        end
    end

    -- Riêng với Gift: Vì Gift không phải là Player, ta vẫn phải quét nhưng quét ít hơn
    -- Tần suất quét Gift: 1 giây 1 lần để chống lag
end)

-- Tách riêng logic quét Gift ra để không làm nặng vòng lặp Render
task.spawn(function()
    while task.wait(1) do -- Quét Gift mỗi giây 1 lần là đủ
        for _, v in pairs(workspace:GetDescendants()) do
            if v.Name:sub(1, #TargetKeyword) == TargetKeyword and v:IsA("BasePart") then
                if not v:FindFirstChild("GiftTag") then
                    -- Tạo Tag cho Gift (giống logic cũ)
                end
            end
        end
    end
end)
