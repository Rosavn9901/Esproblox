        local dist = math.huge
        for _, p in pairs(game.Players:GetPlayers()) do
            if p ~= game.Players.LocalPlayer and p.Character and p.Character:FindFirstChild("HumanoidRootPart") then
                local pos, onScreen = workspace.CurrentCamera:WorldToViewportPoint(p.Character.HumanoidRootPart.Position)
                if onScreen then
                    local mag = (Vector2.new(pos.X, pos.Y) - Vector2.new(workspace.CurrentCamera.ViewportSize.X/2, workspace.CurrentCamera.ViewportSize.Y/2)).Magnitude
                    if mag < dist then target = p; dist = mag end
                end
            end
        end
        if target then
            LockedPlayer = target
            MainBtn.BackgroundColor3 = Color3.fromRGB(180, 0, 0)

-- HÀM HỖ TRỢ & ESP
-- ==========================
local function isValid(plr)
    if not plr or plr == game.Players.LocalPlayer then return false end
    if not plr.Character or not plr.Character:FindFirstChild("HumanoidRootPart") or not plr.Character:FindFirstChild("Humanoid") then return false end
    if _G.TeamCheck and plr.Team == game.Players.LocalPlayer.Team then return false end
    for _, name in pairs(_G.WhitelistedPlayers) do if plr.Name == name then return false end end
    return true
end

local function CreateESP(plr)
    local InfoTag = Drawing.new("Text")
    local Box = Drawing.new("Square")
    local HealthBar = Drawing.new("Square")

    InfoTag.Visible = false; InfoTag.Center = true; InfoTag.Outline = true; InfoTag.Font = 2; InfoTag.Size = 13
    Box.Visible = false; Box.Thickness = 1; Box.Filled = false
    HealthBar.Visible = false; HealthBar.Thickness = 1; HealthBar.Filled = true

    game:GetService("RunService").RenderStepped:Connect(function()
        if plr.Character and plr.Character:FindFirstChild("Humanoid") and isValid(plr) then
            local HRP = plr.Character.HumanoidRootPart
            local Hum = plr.Character.Humanoid
            local pos, onScreen = workspace.CurrentCamera:WorldToViewportPoint(HRP.Position)

            if onScreen then
                local sizeX = 1000 / pos.Z
                local sizeY = 2000 / pos.Z
                local boxPos = Vector2.new(pos.X - sizeX / 2, pos.Y - sizeY / 2)
                local isLocked = (LockedPlayer == plr)

                if _G.ESP_Enabled then
                    local dist = math.floor((game.Players.LocalPlayer.Character.HumanoidRootPart.Position - HRP.Position).Magnitude)
                    InfoTag.Position = Vector2.new(pos.X, pos.Y - (sizeY / 2) - 15)
                    InfoTag.Text = string.format("%s | %d HP | %dm", plr.Name, math.floor(Hum.Health), dist)
                    InfoTag.Color = isLocked and Color3.new(1,0,0) or _G.Color_Name
                    InfoTag.Visible = true
                else InfoTag.Visible = false end

                if _G.Box_Enabled then
                    Box.Size = Vector2.new(sizeX, sizeY); Box.Position = boxPos
                    Box.Color = isLocked and Color3.new(1,0,0) or _G.Color_Box
                    Box.Visible = true
                else Box.Visible = false end

                if _G.HealthBar_Enabled then
                    local healthPct = math.clamp(Hum.Health / Hum.MaxHealth, 0, 1)
                    local barSize = sizeY * healthPct
                    HealthBar.Size = Vector2.new(1.2, barSize)
                    HealthBar.Position = Vector2.new(boxPos.X - 4, boxPos.Y + (sizeY - barSize))
                    HealthBar.Color = Color3.fromHSV(healthPct * 0.3, 0.9, 1)
                    HealthBar.Visible = true
                else HealthBar.Visible = false end
            else InfoTag.Visible = false; Box.Visible = false; HealthBar.Visible = false end
        else
            InfoTag.Visible = false; Box.Visible = false; HealthBar.Visible = false
        end
    end)
end

for _, p in pairs(game.Players:GetPlayers()) do CreateESP(p) end
game.Players.PlayerAdded:Connect(CreateESP)

-- ==========================
-- LOGIC CHIẾN ĐẤU (GHIM TÂM & HƯỚNG MẶT)
-- ==========================
game:GetService("RunService").RenderStepped:Connect(function()
    local Camera = workspace.CurrentCamera
    local MyChar = game.Players.LocalPlayer.Character
    if not MyChar or not MyChar:FindFirstChild("HumanoidRootPart") then return end
    
    local finalTarget = nil
    
    -- 1. Xác định mục tiêu (Ưu tiên nút L)
    if _G.AimButton_Enabled and LockedPlayer and LockedPlayer.Character and LockedPlayer.Character:FindFirstChild("HumanoidRootPart") then
        finalTarget = LockedPlayer.Character.HumanoidRootPart
    elseif _G.Aimlock_Enabled then
        local targetDist = math.huge
        for _, p in pairs(game.Players:GetPlayers()) do
            if isValid(p) then
                local pos, onScreen = Camera:WorldToViewportPoint(p.Character.HumanoidRootPart.Position)
                if onScreen then
                    local mag = (Vector2.new(pos.X, pos.Y) - Vector2.new(Camera.ViewportSize.X/2, Camera.ViewportSize.Y/2)).Magnitude
                    if mag < targetDist then
                        finalTarget = p.Character.HumanoidRootPart
                        targetDist = mag
                    end
                end
            end
        end
    end

    -- 2. Thực thi Ghim tâm và Xoay người
    if finalTarget then
        local tPos = finalTarget.Position
        
        -- Ghim Camera
        if _G.HardLock then
            Camera.CFrame = CFrame.new(Camera.CFrame.Position, tPos)
        else
            Camera.CFrame = Camera.CFrame:Lerp(CFrame.new(Camera.CFrame.Position, tPos), 0.15)
        end
        
        -- XOAY NHÂN VẬT ĐỂ ĐỒNG BỘ HƯỚNG CHẠY (QUAN TRỌNG)
        local myHRP = MyChar.HumanoidRootPart
        myHRP.CFrame = CFrame.new(myHRP.Position, Vector3.new(tPos.X, myHRP.Position.Y, tPos.Z))
    end

    -- 3. Hitbox Logic
    if _G.HitboxEnabled then
        for _, p in pairs(game.Players:GetPlayers()) do
            if isValid(p) then
                p.Character.HumanoidRootPart.Size = Vector3.new(_G.HitboxSize, _G.HitboxSize, _G.HitboxSize)
                p.Character.HumanoidRootPart.Transparency = 0.7
            end
        end
    end
end)
